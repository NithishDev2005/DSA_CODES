import java.util.Scanner;

public class bananas{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int k=sc.nextInt();
        int n=sc.nextInt();
        int w=sc.nextInt();
        
        int cost=k*w*(w+1)/2;
        int bo=Math.max(0,cost-n);
        
        System.out.println(bo);
    }
}
