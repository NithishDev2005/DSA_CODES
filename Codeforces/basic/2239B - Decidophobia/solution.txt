import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int d = sc.nextInt();

            long[] a = new long[3 * n];
            long[] prefix = new long[3 * n + 1];

            // Read array and repeat it 3 times
            for (int i = 0; i < n; i++) {
                a[i] = sc.nextLong();
                a[i + n] = a[i];
                a[i + 2 * n] = a[i];
            }

            // Prefix sum
            for (int i = 0; i < 3 * n; i++) {
                prefix[i + 1] = prefix[i] + a[i];
            }

            long ans = 0;

            for (int i = 0; i < n; i++) {

                // Use middle copy to handle circular array
                int pos = i + n;

                // Sum of d people on the left
                long left =
                        prefix[pos] - prefix[pos - d];

                // Sum of d people on the right
                long right =
                        prefix[pos + d + 1] - prefix[pos + 1];

                // Benefit if person i receives a gift
                long gain =
                        2L * d * a[i] - left - right;

                if (gain > 0) {
                    ans += gain;
                }
            }

            System.out.println(ans);
        }

        sc.close();
    }
}
