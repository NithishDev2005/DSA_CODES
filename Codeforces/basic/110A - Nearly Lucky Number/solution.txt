import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        
        if(sc.hasNext()){
            String n=sc.next();
            int lucky=0;
            
            for(int i=0;i<n.length();i++){
                char c=n.charAt(i);
                if(c=='4'|| c=='7'){
                    lucky++;
                }
            }
            
            if(lucky==4 || lucky==7){
                System.out.println("YES");
            }
            else{
                System.out.println("NO");
            }
        }
    }
}
