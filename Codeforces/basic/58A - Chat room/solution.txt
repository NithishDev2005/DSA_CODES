import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        
        String s=sc.next();
        String target="hello";
        int ind=0;
        
        for(char c:s.toCharArray()){
            if(ind<target.length() && c==target.charAt(ind)){
                ind++;
            }
        }
        
        if(ind==target.length()){
            System.out.println("YES");
        }
        else{
            System.out.println("NO");
        }
    }
}
