import java.util.Scanner;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        
        if(sc.hasNextInt()){
            int n=sc.nextInt();
            int k=sc.nextInt();
            
            int[] scores=new int[n];
            for(int i=0;i<n;i++){
                scores[i]=sc.nextInt();
            }
            
            int kthscore=scores[k-1];
            
            int adva=0;
            
            for(int i=0;i<n;i++){
                if(scores[i]>=kthscore&&scores[i]>0){
                    adva++;
                }
                else{
                    break;
                }
            }
            System.out.println(adva);
        }
    }
}
