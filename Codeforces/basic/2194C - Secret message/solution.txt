import java.util.*;
public class Main {
    static void solve(Scanner sc) {
        int n = sc.nextInt();
        int k = sc.nextInt();
        String[] s = new String[k];
        for (int i = 0; i < k; i++) {
            s[i] = sc.next();
        }
        int[] m = new int[n];
        for (int j = 0; j < n; j++) {
            int mask = 0;
            for (int i = 0; i < k; i++) {
                int bit = s[i].charAt(j) - 'a';
                mask |= (1 << bit);
            }
            m[j] = mask;
        }

        
        ArrayList<Integer> v = new ArrayList<>();
        for (int i = 1; i * i <= n; i++) {
            if (n % i == 0) {
                v.add(i);
                if (i * i != n) v.add(n / i);
            }
        }
        Collections.sort(v);

        for (int d : v) {
            StringBuilder a = new StringBuilder();
            boolean ok = true;
            for (int i = 0; i < d; i++) {
                int x = (1 << 26) - 1;
                for (int j = i; j < n; j += d) {
                    x &= m[j];
                }
                if (x == 0) {
                    ok = false;
                    break;
                }
                int b = 0;
                while (((x >> b) & 1) == 0) b++;
                a.append((char) ('a' + b));
            }

            if (ok) {
                int repeat = n / d;
                for (int i = 0; i < repeat; i++) System.out.print(a);
                System.out.println();
                return;
            }
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();
        while (t-- > 0) {
            solve(sc);
        }
    }
}
