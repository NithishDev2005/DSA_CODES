import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        String s=sc.next();
        
        boolean danger=false;
        int cnt=1;
        
        for(int i=1;i<s.length();i++){
            if(s.charAt(i)==s.charAt(i-1)){
                cnt++;
                
                if(cnt>=7){
                    danger=true;
                    
                }
            }
                else{
                    cnt=1;
                
            }
        }
        if(danger){
            System.out.println("YES");
        }
        else{
            System.out.println("NO");
        }
    }
}
